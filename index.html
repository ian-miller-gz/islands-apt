Islands apt

sudo curl -fsSL https://ian-miller-gz.github.io/islands-packages/islands.gpg -o /usr/share/keyrings/islands.gpg
echo "deb [signed-by=/usr/share/keyrings/islands.gpg arch=amd64] https://ian-miller-gz.github.io/islands-packages stable main" | sudo tee /etc/apt/sources.list.d/islands.list
sudo apt update
sudo apt install islands

Islands pacman

curl -fsSL https://ian-miller-gz.github.io/islands-packages/islands.gpg -o /tmp/islands.gpg
sudo pacman-key --add /tmp/islands.gpg
sudo pacman-key --lsign-key CA2AC1FB65D6629F
printf '[islands]\nSigLevel = Required DatabaseOptional\nServer = https://ian-miller-gz.github.io/islands-packages/arch\n' | sudo tee -a /etc/pacman.conf
sudo pacman -Sy islands
